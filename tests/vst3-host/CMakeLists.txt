cmake_minimum_required(VERSION 3.22)
project(Vst3RoutingCheck LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory("../../third_party/JUCE" "${CMAKE_BINARY_DIR}/JUCE" EXCLUDE_FROM_ALL)
add_executable(vst3_routing_check ../Vst3HostTests.cpp)
target_compile_definitions(vst3_routing_check PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(vst3_routing_check PRIVATE juce::juce_audio_processors juce::juce_recommended_config_flags)
if(MSVC)
    target_compile_options(vst3_routing_check PRIVATE /MP4)
endif()
